    <script>
        // Dữ liệu câu hỏi
        const situations = [
            { id: 1, text: "1. Nhận được tệp lạ từ người không quen." },
            { id: 2, text: "2. Máy tính xuất hiện cảnh báo có virus." },
            { id: 3, text: "3. Bạn rủ dùng mật khẩu là ngày sinh." },
            { id: 4, text: "4. Muốn sao lưu bài tập quan trọng." }
        ];

        const solutions = [
            { matchId: 3, text: "Từ chối và chọn mật khẩu khó đoán hơn." },
            { matchId: 2, text: "Cập nhật và sử dụng phần mềm diệt virus." },
            { matchId: 4, text: "Lưu vào USB, ổ cứng ngoài hoặc Google Drive." }, 
            { matchId: 1, text: "Không mở; xóa hoặc báo cho người lớn/giáo viên." }
        ];

        let selectedSituation = null;
        let selectedSolution = null;
        let correctPairs = 0;
        let audioInitialized = false;

        // Bộ tổng hợp âm thanh (Synthesizers)
        let synthCorrect;
        let synthWrong;
        let synthWin;

        // Khởi tạo âm thanh khi người dùng tương tác lần đầu
        async function initAudio() {
            if (!audioInitialized) {
                await Tone.start();
                
                // Âm thanh khi đúng (Tiếng Ping cao)
                synthCorrect = new Tone.Synth({
                    oscillator: { type: "sine" },
                    envelope: { attack: 0.01, decay: 0.1, sustain: 0, release: 0.1 }
                }).toDestination();
                synthCorrect.volume.value = -10;

                // Âm thanh khi sai (Tiếng Buzz trầm)
                synthWrong = new Tone.MembraneSynth().toDestination();
                synthWrong.volume.value = -5;

                // Âm thanh khi chiến thắng
                synthWin = new Tone.PolySynth(Tone.Synth).toDestination();
                synthWin.volume.value = -10;
                
                audioInitialized = true;
            }
        }

        function playWrongSound() {
            if (audioInitialized && synthWrong) {
                synthWrong.triggerAttackRelease("C2", "8n");
            }
        }

        function playCorrectSound() {
            if (audioInitialized && synthCorrect) {
                synthCorrect.triggerAttackRelease("E5", "8n");
            }
        }

        function playWinSound() {
            if (audioInitialized && synthWin) {
                const now = Tone.now();
                synthWin.triggerAttackRelease(["C4", "E4", "G4"], "8n", now);
                synthWin.triggerAttackRelease(["C4", "E4", "G4", "C5"], "4n", now + 0.15);
            }
        }

        // Hàm render danh sách ra màn hình
        function renderList(containerId, data, isSituation) {
            const container = document.getElementById(containerId);
            container.innerHTML = ''; // Xóa nội dung cũ
            
            const labels = ['A', 'B', 'C', 'D']; // Danh sách nhãn cố định cho cột Cách xử lí
            
            data.forEach((item, index) => {
                const div = document.createElement('div');
                div.className = 'item-card p-4 border-2 border-slate-200 rounded-xl cursor-pointer bg-slate-50 hover:bg-slate-100 text-slate-700 font-medium leading-relaxed shadow-sm text-sm sm:text-base';
                
                // Gắn nhãn A, B, C, D động dựa trên vị trí sau khi đã đảo lộn
                if (isSituation) {
                    div.textContent = item.text;
                } else {
                    div.textContent = `${labels[index]}. ${item.text}`;
                }
                
                div.dataset.id = isSituation ? item.id : item.matchId;
                div.onclick = async () => {
                    await initAudio(); // Đảm bảo âm thanh được kích hoạt
                    handleSelect(div, isSituation);
                };
                container.appendChild(div);
            });
        }

        function handleSelect(element, isSituation) {
            if (element.classList.contains('correct') || element.classList.contains('disabled')) return;

            if (isSituation) {
                if (selectedSituation) selectedSituation.classList.remove('selected', 'incorrect');
                selectedSituation = element;
            } else {
                if (selectedSolution) selectedSolution.classList.remove('selected', 'incorrect');
                selectedSolution = element;
            }

            element.classList.add('selected');
            element.classList.remove('incorrect');

            if (selectedSituation && selectedSolution) {
                checkMatch();
            }
        }

        function checkMatch() {
            if (selectedSituation.dataset.id === selectedSolution.dataset.id) {
                // Trả lời đúng
                selectedSituation.classList.replace('selected', 'correct');
                selectedSolution.classList.replace('selected', 'correct');
                selectedSituation.classList.add('disabled');
                selectedSolution.classList.add('disabled');
                correctPairs++;
                
                if (correctPairs === situations.length) {
                    playWinSound();
                    const msg = document.getElementById('result-message');
                    msg.textContent = "🎉 Tuyệt vời! Em đã hoàn thành xuất sắc bài tập.";
                    msg.classList.remove('opacity-0');
                    msg.classList.add('opacity-100');
                    
                    // Hiện nút làm lại
                    document.getElementById('reset-btn').classList.remove('hidden');
                } else {
                    playCorrectSound();
                }
            } else {
                // Trả lời sai
                playWrongSound();
                selectedSituation.classList.replace('selected', 'incorrect');
                selectedSolution.classList.replace('selected', 'incorrect');
                
                const tempSit = selectedSituation;
                const tempSol = selectedSolution;
                
                // Bỏ hiệu ứng rung lắc sau 0.6s
                setTimeout(() => {
                    tempSit.classList.remove('incorrect', 'selected');
                    tempSol.classList.remove('incorrect', 'selected');
                }, 600);
            }
            selectedSituation = null;
            selectedSolution = null;
        }

        // Hàm làm lại bài tập
        function resetGame() {
            selectedSituation = null;
            selectedSolution = null;
            correctPairs = 0;
            
            // Ẩn thông báo và nút làm lại
            const msg = document.getElementById('result-message');
            msg.classList.remove('opacity-100');
            msg.classList.add('opacity-0');
            document.getElementById('reset-btn').classList.add('hidden');
            
            // Khởi tạo lại giao diện
            renderList('col-situations', situations, true);
            renderList('col-solutions', shuffle([...solutions]), false);
        }

        // Hàm đảo ngẫu nhiên thứ tự cột đáp án
        function shuffle(array) {
            for (let i = array.length - 1; i > 0; i--) {
                const j = Math.floor(Math.random() * (i + 1));
                [array[i], array[j]] = [array[j], array[i]];
            }
            return array;
        }

        // Khởi chạy lần đầu
        renderList('col-situations', situations, true);
        renderList('col-solutions', shuffle([...solutions]), false);
    </script>
